# Fresh independent profile. Extra flags: -brawlplay -brawlnocut.
# Explicit travel avoids depending on the project's default day/title map.
open /Game/Brawl/Maps/L_RustyAnchor
waitfor FTBRAWL start THE RUSTY ANCHOR|25
waitfor FT BRAWL SAVE result=ok reason=EncounterEntered kind=night level=0 stage=0|10
ftb.SessionState
ft.SaveDelayMs 2000
ft.SaveFail 1
mark
ftb.Win 1
waitfor FT BRAWL SAVE queued reason=BossDefeatedTest kind=nightoutro level=0 stage=2|5
# A close request queues immediately while the real worker is still pending.
ftb.WindowClose
waitfor FT BRAWL SYSTEM screen=7|5
ftb.SessionState
shot brawl-async-saving-queued
# Cancel is responsive; it cancels exit intent, while the save remains in flight.
press Escape
waitfor FT BRAWL SYSTEM screen=2|5
ftb.SessionState
waitfor FT BRAWL SAVE result=failed reason=BossDefeatedTest kind=nightoutro level=0 stage=2|8
ftb.SessionState
# Re-open the error panel after a short resume. Continue must still describe wave1.
press Escape
wait 0.8
press Escape
waitfor FT BRAWL SYSTEM screen=6|5
ftb.SessionState
shot brawl-async-failed-durable-wave
ft.SaveFail 0
mark
press Enter
waitfor FT BRAWL SAVE queued reason=Exit kind=nightoutro level=0 stage=2|5
waitfor FT BRAWL SYSTEM screen=7|5
ftb.SessionState
shot brawl-async-retry-saving
# Queue another close during the successful retry. It must not exit early.
ftb.WindowClose
waitfor FT BRAWL SYSTEM exit queued=desktop|5
waitfor FT BRAWL SAVE result=ok reason=Exit kind=nightoutro level=0 stage=2|8
waitfor FT BRAWL SYSTEM screen=5|5
ftb.SessionState
shot brawl-async-durable-outro
press Down
press Enter
